    SELECT
        source_id, ra, dec, phot_g_mean_mag,

        gaiadr3.gaia_source 
    WHERE

    url,

    table_selector="table"
):
    """
    Generic scraper for column-definition tables.

    Parameters
    ----------
    url : str
        URL of documentation page
    col_map : dict
        Mapping of logical field → cell index
        Example:
        {
            "db_col": 0,
            "label": 1,
            "description": 2
        }
    table_selector : str

            db_col_name = cells[col_map["db_col"]].get_text(strip=True)
            table_label = cells[col_map["label"]].get_text(strip=True)
